public class SubsequenceGeneration {
	public static void main(String[] args) {
		generateSubsequence("abc", "", 0);
		System.out.println("+++++++++++++++++");
		generateSubsequenceUsingForLoop("abc", "", 0, 2);
	}

	public static void generateSubsequence(String str, String ans, int i) {
		if (i == str.length()) {
			System.out.println(ans + "=");
			return;
		}
		generateSubsequence(str, ans + str.charAt(i), i + 1);
		generateSubsequence(str, ans, i + 1);
	}
	
	public static void generateSubsequenceUsingForLoop(String str, String ans,int l, int h) {
		System.out.println(ans+"=");
		for(int i=l;i<=h;i++) {
			generateSubsequenceUsingForLoop(str, ans+str.charAt(i), i+1, h);
		}
	}
}

class Solution {
    public List<List<Integer>> subsets(int[] nums) {
        List<List<Integer>> ans = new ArrayList<>();
        subseq(nums,0, new ArrayList<Integer>(), ans );
        return ans;
    }

    public void subseq(int[] nums, int index, List<Integer> curr, List<List<Integer>> ans){
        if(index==nums.length){
            ans.add(new ArrayList<>(curr));
            return; 
        }
        curr.add(nums[index]);
        subseq(nums, index+1, curr, ans);
        curr.removeLast();
        subseq(nums, index+1, curr, ans);
    }
}

class Solution {
    public List<List<Integer>> subsetsWithDup(int[] nums) {
        List<List<Integer>> set  = new ArrayList<>();
        Arrays.sort(nums);
        gs(nums, set, new ArrayList<Integer>(), 0);
        return set;
    }
    public void gs(int[] nums, List<List<Integer>> set, List<Integer> list, int i){
        if(i==nums.length){
            set.add(new ArrayList<>(list));
            return;
        }
        list.add(nums[i]);
        gs(nums,set, list, i+1);
        list.removeLast();
        while(i+1<nums.length && nums[i]==nums[i+1]){
            i++;
        }
        gs(nums,set, list,i+1);
    }

}


Backtracking:-
=============

- Backtracking is an algorithmic technique in which we make a choice, explore that choice recursively, and if it does not lead to a valid solution, we undo that choice and try another possibility.

In one line:-
	Backtracking = Recursion + Choice + Undo


Real Life Example
----------------------
Suppose you are in a maze:-

	S . .
	# # .
	. . E

You have 4 choices:-

	Up
	Down
	Left
	Right

	You choose left.

	↓

	Dead End.

What you will do now?

You came back

	↓

try another path

- That process is called as Backtracking.

- Why is it called Backtracking?

- Backtracking uses two words :-

	Back

	+

	Tracking

Means:-

Go Forward

↓

Wrong Path

↓

Come Back

↓

Try Another Path

General Steps of Backtracking
-------------------------------
Every Backtracking problem follows following pattern to solve questions:-

1. Choose

↓

2. Check

↓

3. Explore

↓

4. Undo

↓

5. Try Next Choice

Simple Backtracking example:-
-----------------------------
Code:-
-----
public class SimpleBacktrackingExample {
	
	public static void main(String[] args) {
		path(0,9);
	}
	
	public static boolean path(int currentPosition, int targetPosition) {
		if(currentPosition==targetPosition) {
			System.out.println("We have reached at destination "+targetPosition);
			return true;
		}
		
		if(currentPosition>targetPosition) {
			System.out.println("We reached at Dead End at position "+currentPosition);
			return false;
		}
		
		System.out.println("We are taking 2 step at "+currentPosition);
		if(path(currentPosition+2,targetPosition)) {
			return true;
		}
		
		System.out.println("We have not found our destination so we need to backtrack from here position is "+ currentPosition+" and we will take 1 step");
		if(path(currentPosition+1, targetPosition)) {
			return true;
		}
		return false; 
	}

}

We have reached at destination 9

class Solution {
    public List<List<Integer>> permute(int[] nums) {    
        List<List<Integer>> ans = new ArrayList<>();
        generatePermutations(nums,ans,0, nums.length-1);
        return ans;    
    }


    public void generatePermutations(int[] nums, List<List<Integer>> ans, int low, int high){
        if(low==high){
            ArrayList<Integer> li = new ArrayList<>();
            for(int x : nums){
                li.add(x);
            }
            ans.add(li);
        }else{
            for(int i=low; i<=high; i++){
                swap(nums,low,i);
                generatePermutations(nums, ans, low+1, high);
                swap(nums,i, low);
            }
        }
    }
    public void swap(int []nums, int i, int j){
        int temp = nums[i];
        nums[i] = nums[j];
        nums[j] = temp;
    }
}

class Solution {
    public ArrayList<String> ratInMaze(int[][] maze) {
        // code here
        ArrayList<String> ans = new ArrayList<>();
        RIM(maze, 0, 0, ans, "");
        return ans;
    }
    public void RIM(int[][]maze, int row, int col, ArrayList<String> ans, String curr){
        //All the places where we can not go
        if(row<0 || col<0 || row>=maze.length || col>=maze.length || maze[row][col]==0 || maze[row][col]==-1){            
		return;
        }
        //Reached at destination
        if(row==maze.length-1 && col == maze.length-1){
            ans.add(curr);
            return;
        }
        maze[row][col] = -1;//Mark visited that place at that time
        
        RIM(maze, row+1, col, ans, curr+"D");//Down
        RIM(maze, row, col-1, ans, curr+"L");//Left
        RIM(maze, row, col+1, ans, curr+"R");//Right
        RIM(maze, row-1, col, ans, curr+"U");//Up
 
        maze[row][col] = 1;//When I explore that place I need to backtrack for other options or Mark not visited
    }

}

class Solution {
    public List<List<String>> solveNQueens(int n) {
        char[][]chess = new char[n][n];
        for(int i=0; i<n; i++){
            for(int j=0; j<n; j++){
                chess[i][j]='.';
            }
        }
        List<List<String>> ans = new ArrayList<>();
        placeQueens(chess, 0, ans);
        return ans;
    }

    public void placeQueens(char[][]chess,int row, List<List<String>> ans){
        if(row==chess.length){
            List<String> list = new ArrayList<>();
            for(int i=0; i<chess.length; i++){
                list.add(new String(chess[i]));
            } 
            ans.add(list);
            return;
        }
        for(int col =0; col<chess.length; col++){
            if(isQueenSafe(chess, row, col)){
                chess[row][col] = 'Q';
                placeQueens(chess, row+1, ans);
                chess[row][col] = '.';
            }
        }
    }
    public boolean isQueenSafe(char[][]chess, int row, int col){
        // Vertical Check
        for(int i=row-1; i>=0; i--){
            if(chess[i][col]=='Q') return false;
        }

        // Left Diagonal
        int i = row-1;
        int j = col-1;
        while(i>=0 && j>=0){
            if(chess[i][j]=='Q') return false;
            i--;
            j--;
        }
        // Right Diagonal
        i = row-1;
        j = col+1;
        while(i>=0 && j<chess.length){
            if(chess[i][j]=='Q') return false;
            i--;
            j++;
        }
        return true;
    }
}

class Solution {
    public void solveSudoku(char[][] board) {
        solving(board, 0, 0);
    }
    public boolean solving(char[][] board, int row, int col){
        if(row==9){
            // System.out.println(Arrays.deepToString(board));
            return true;
        }
        int nRow =0;
        int nCol = 0;
        if(col == 8){
            nRow = row+1;
            nCol = 0;
        }else{
            nCol = col+1;
            nRow = row;
        }
        if(board[row][col] !='.'){
            return solving(board, nRow, nCol);
        }else{
            for(int i=1; i<=9; i++){
                if(isValid(board, row, col, i)){
                    char ch = (char)(i + '0');
                    board[row][col] = ch;
                    if(solving(board, nRow, nCol)){
                        return true;
                    }
                    board[row][col] = '.';
                }
            }
        }
        return false;

    }
    public boolean isValid(char[][]board, int row, int col, int target){
        // target int converting into char
        char ch = (char) (target + '0');
        // Checking Row
        for(int i=0; i<9; i++){
            if(board[i][col] == ch){
                return false;
            }
        }
        // Checking column
        for(int i=0; i<9; i++){
            if(board[row][i]==ch){
                return false;
            }
        }
        // checking that particular 3x3 Matrix/grid
        int gridRow  = (row/3)*3;
        int gridCol = (col/3)*3;
        for(int i=0; i<3; i++){
            for(int j=0; j<3; j++){
                if(board[i+gridRow][j+gridCol]==ch){
                    return false;
                }
            }
        }
        return true;

    }
}

class Solution {
    public List<List<Integer>> combine(int n, int k) {
        List<List<Integer>> ans=new ArrayList<>();
        backtrack(1,n,k,new ArrayList<>(),ans);
        return ans;
        
    }
    public void backtrack(int start,int n,int k,List<Integer>curr,List<List<Integer>> ans){
        if(curr.size()==k){
            // System.out.println(curr);
            ans.add(new ArrayList<>(curr));
            return;
        }
        for(int i=start;i<=n;i++){
            curr.add(i);
            backtrack(i+1,n,k,curr,ans);
            curr.remove(curr.size()-1);
        }
        
    }
}

class Solution {
    public List<List<Integer>> combinationSum2(int[] candidates, int target) {
        Arrays.sort(candidates);
        List<List<Integer>> ans = new ArrayList<>();

        generateComb(candidates, 0, ans, new ArrayList<>(),target);
        return new ArrayList<>(ans);
    }

    public void generateComb(int[] candidates, int index, List<List<Integer>> ans, List<Integer> curr, int target){

        if(target==0){
            ans.add(new ArrayList<>(curr));
            return;
        }
        if(target<0 || index == candidates.length){
            return;
        }
        curr.add(candidates[index]);
        generateComb(candidates, index+1, ans, curr, target-candidates[index]);
        curr.removeLast();
        while(index+1<candidates.length && candidates[index]==candidates[index+1]){
            index++;
        }
        generateComb(candidates, index+1, ans, curr, target);

    }
}

class Solution {
    public List<String> generateParenthesis(int n) {
        List<String> ans = new ArrayList<>();
        generateP(n, 0, 0, new StringBuilder(), ans);
        return ans;
    }
    public void generateP(int n, int open, int close, StringBuilder curr, List<String> ans){
        if(n==open && n == close){
            ans.add(curr.toString());
            return;
        }
        if(open<n){
            curr.append("(");
            generateP(n, open+1, close, curr, ans);
            curr.deleteCharAt(curr.length()-1);
        }
        if(close<open){
            curr.append(")");
            generateP(n, open, close+1, curr, ans);
            curr.deleteCharAt(curr.length()-1);
        }
    }
}

class Result {
  static int countWord(char board[][], String word, int m, int n) {
    int count =0;
      for(int i=0; i<m; i++){
          for(int j=0; j<n; j++){
              count += counting(board, i, j, word, 0);
          }
      }
      return count;
  }
  static int counting(char board[][], int row, int col, String word, int index){
      int count =0;
      if(row<0 || col<0 || row==board.length || col==board[0].length || board[row][col] != word.charAt(index) || 
         board[row][col]=='#'){
          return 0;
      }
      if(index == word.length()-1){
          return 1;
      }
      char temp = board[row][col];
      board[row][col] = '#';
      //Up 
      count += counting(board, row-1, col, word, index+1);
      //Down
       count += counting(board, row+1, col, word, index+1);
      //Left
       count += counting(board, row, col-1, word, index+1);
      //Right
       count += counting(board, row, col+1, word, index+1);
      board[row][col] = temp; 
      return count;
      
  }
}
